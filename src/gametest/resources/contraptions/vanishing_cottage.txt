# The vanishing cottage (VanishingCottageGameTests, examples station 29): two hidden area shifters take a whole
# cottage out of the visible world, roof first, and build it back from the floor. Needs the game rule
# phasecraft:area_shifters on (off by default). Check with: python3 tools/redstone_check.py <this file>
#
# The cottage is the box x 3..9, y 1..5, z 3..9: a plank floor, plank walls with log corners, a doorway north
# (z = 3), glass windows, a spruce roof, and inside a chest, a crafting table and a furnace. The two area shifters
# sit at its opposite corners (3 1 3, the leader, and 9 5 9) with a delay of 3: a layer every 3 redstone ticks.
# The lever stands on the ground north of the doorway; a phase port under it carries its power into the hidden
# layer, and two hidden wires take it to the corners: west and south to the low one, east and up a hidden stair
# east of the cottage to the high one (a repeater halfway).
#
# Only 13 x 12 is used: the area is 24 x 24 so that a GameTest beside it is more than 15 blocks away, out of the
# corners' reach (they pair with their nearest other area shifter within 15).
#
# Signals (@name, @out<in):
#   L   the lever, through the port
#   R   L again, after the repeater on the way up
size 24 7 24
lever 6 1 0
fill visible 0 0 0 12 0 11 minecraft:grass_block
# A foundation the cottage stands on, left when it goes (grass under it would turn to dirt and leave a patch).
fill visible 2 0 2 10 0 10 minecraft:stone_bricks
# The cottage.
fill visible 3 1 3 9 1 9 minecraft:oak_planks
fill visible 3 2 3 9 4 9 minecraft:oak_planks
fill visible 4 2 4 8 4 8 minecraft:air
fill visible 3 2 3 3 4 3 minecraft:oak_log
fill visible 9 2 3 9 4 3 minecraft:oak_log
fill visible 3 2 9 3 4 9 minecraft:oak_log
fill visible 9 2 9 9 4 9 minecraft:oak_log
fill visible 6 2 3 6 3 3 minecraft:air
visible 3 3 6 minecraft:glass
visible 9 3 6 minecraft:glass
visible 5 3 9 minecraft:glass
visible 7 3 9 minecraft:glass
fill visible 3 5 3 9 5 9 minecraft:spruce_planks
visible 7 2 8 minecraft:chest[facing=north]
visible 5 2 8 minecraft:crafting_table
visible 4 2 8 minecraft:furnace[facing=north]
items visible 7 2 8 minecraft:bread*6 minecraft:apple*3 minecraft:iron_ingot*5
# The lever, and the port under it.
visible 6 1 0 minecraft:lever[face=floor,facing=north,powered=false]
hidden 6 0 0 phasecraft:phase_port @L
# To the low corner: west along z = 0, then south.
fill hidden 3 0 0 5 0 0 minecraft:stone
fill hidden 3 0 1 3 0 2 minecraft:stone
hidden 3 1 0 minecraft:redstone_wire @L
hidden 4 1 0 minecraft:redstone_wire @L
hidden 5 1 0 minecraft:redstone_wire @L
hidden 6 1 0 minecraft:redstone_wire @L
hidden 3 1 1 minecraft:redstone_wire @L
hidden 3 1 2 minecraft:redstone_wire @L
# To the high corner: east along z = 0, south up a stair at x = 11, a repeater, then along y = 5 and west into it.
fill hidden 7 0 0 11 0 0 minecraft:stone
hidden 11 0 1 minecraft:stone
hidden 11 1 2 minecraft:stone
hidden 11 2 3 minecraft:stone
hidden 11 3 4 minecraft:stone
fill hidden 11 4 5 11 4 9 minecraft:stone
hidden 10 4 9 minecraft:stone
hidden 7 1 0 minecraft:redstone_wire @L
hidden 8 1 0 minecraft:redstone_wire @L
hidden 9 1 0 minecraft:redstone_wire @L
hidden 10 1 0 minecraft:redstone_wire @L
hidden 11 1 0 minecraft:redstone_wire @L
hidden 11 1 1 minecraft:redstone_wire @L
hidden 11 2 2 minecraft:redstone_wire @L
hidden 11 3 3 minecraft:redstone_wire @L
hidden 11 4 4 minecraft:redstone_wire @L
hidden 11 5 5 minecraft:redstone_wire @L
hidden 11 5 6 minecraft:repeater[facing=north,delay=1,powered=false] @R<L
hidden 11 5 7 minecraft:redstone_wire @R
hidden 11 5 8 minecraft:redstone_wire @R
hidden 11 5 9 minecraft:redstone_wire @R
hidden 10 5 9 minecraft:redstone_wire @R
# The corners last.
hidden 3 1 3 phasecraft:area_shifter[delay=3] @L
hidden 9 5 9 phasecraft:area_shifter[delay=3] @R
